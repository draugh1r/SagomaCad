cmake_minimum_required(VERSION 3.25)
project(sagomacad VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()

find_package(SDL3 CONFIG REQUIRED)
find_package(imgui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(lunasvg CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(OpenCASCADE CONFIG REQUIRED)
find_path(STB_INCLUDE_DIRS stb_image_write.h PATH_SUFFIXES stb REQUIRED)

foreach(module core solver)
  add_library(sagomacad_${module} INTERFACE)
  target_include_directories(sagomacad_${module} INTERFACE ${PROJECT_SOURCE_DIR}/src)
endforeach()
add_library(sagomacad_doc STATIC src/doc/document.cpp src/doc/expression.cpp)
target_include_directories(sagomacad_doc PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(sagomacad_doc PUBLIC sagomacad_core nlohmann_json::nlohmann_json)
add_library(sagomacad_ui STATIC src/ui/toolbar_catalog.cpp src/ui/parameter_editor.cpp src/ui/camera.cpp)
target_include_directories(sagomacad_ui PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(sagomacad_solver INTERFACE sagomacad_core sagomacad_doc)
add_library(sagomacad_kernel STATIC src/kernel/box_volume.cpp src/kernel/shape.cpp)
target_include_directories(sagomacad_kernel PUBLIC ${PROJECT_SOURCE_DIR}/src ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(sagomacad_kernel PUBLIC sagomacad_core sagomacad_doc ${OpenCASCADE_LIBRARIES})
add_library(sagomacad_regen STATIC src/regen/regenerator.cpp)
target_include_directories(sagomacad_regen PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(sagomacad_regen PUBLIC sagomacad_core sagomacad_doc sagomacad_solver sagomacad_kernel)
add_library(sagomacad_mesh STATIC src/mesh/surface.cpp)
target_include_directories(sagomacad_mesh PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(sagomacad_mesh PUBLIC sagomacad_core sagomacad_doc sagomacad_kernel)
add_library(sagomacad_engine STATIC src/engine/registry.cpp src/engine/cmds/document_cmds.cpp
    src/engine/cmds/param_cmds.cpp src/engine/cmds/edit_cmds.cpp src/engine/cmds/primitive_cmds.cpp src/engine/cmds/export_cmds.cpp)
target_include_directories(sagomacad_engine PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(sagomacad_engine PUBLIC sagomacad_core sagomacad_doc sagomacad_solver sagomacad_kernel sagomacad_regen sagomacad_mesh)
add_library(sagomacad_automation STATIC src/automation/dispatch.cpp
  src/automation/control_client.cpp src/automation/session_file.cpp)
target_include_directories(sagomacad_automation PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(sagomacad_automation PUBLIC sagomacad_engine nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(sagomacad_automation PRIVATE ws2_32)
endif()
target_link_libraries(sagomacad_ui PUBLIC sagomacad_engine)

add_library(sagomacad_shell STATIC apps/common/shell.cpp apps/common/control.cpp)
target_include_directories(sagomacad_shell PUBLIC ${PROJECT_SOURCE_DIR}/apps/common PRIVATE ${STB_INCLUDE_DIRS})
file(COPY ${PROJECT_SOURCE_DIR}/assets/fonts ${PROJECT_SOURCE_DIR}/assets/icons DESTINATION ${CMAKE_BINARY_DIR}/assets)
target_compile_definitions(sagomacad_shell PRIVATE SAGOMACAD_ASSET_DIR="${CMAKE_BINARY_DIR}/assets")
target_link_libraries(sagomacad_shell PUBLIC sagomacad_ui sagomacad_automation SDL3::SDL3 imgui::imgui OpenGL::GL lunasvg::lunasvg nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(sagomacad_shell PRIVATE ws2_32)
endif()
add_executable(sagomacad apps/sagomacad/main.cpp)
target_link_libraries(sagomacad PRIVATE sagomacad_shell)
add_executable(sagomacad-cli apps/sagomacad-cli/main.cpp apps/sagomacad-cli/fuzz.cpp
  apps/sagomacad-cli/client.cpp)
target_link_libraries(sagomacad-cli PRIVATE sagomacad_shell)
set_target_properties(sagomacad PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/apps/sagomacad$<0:>")
set_target_properties(sagomacad-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/apps/sagomacad-cli$<0:>")

enable_testing()

add_test(NAME layers COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/check_layers.py)

add_test(NAME snapshot COMMAND sagomacad-cli snapshot --size 640x400 --out ${CMAKE_BINARY_DIR}/m0-test.png)
add_test(NAME snapshot_glyphs COMMAND sagomacad-cli snapshot --size 1280x720 --out ${CMAKE_BINARY_DIR}/m0-glyphs.png --glyph-test)
add_test(NAME control COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_control.py $<TARGET_FILE:sagomacad>)
add_executable(kernel_box_test tests/kernel_box.cpp)
target_link_libraries(kernel_box_test PRIVATE sagomacad_kernel)
add_test(NAME kernel_box COMMAND kernel_box_test)
add_executable(kernel_shape_test tests/kernel_shape.cpp)
target_link_libraries(kernel_shape_test PRIVATE sagomacad_kernel)
add_test(NAME kernel_shape COMMAND kernel_shape_test)
add_executable(document_test tests/document.cpp)
target_link_libraries(document_test PRIVATE sagomacad_doc)
add_test(NAME document COMMAND document_test)
add_executable(expression_test tests/expression.cpp)
target_link_libraries(expression_test PRIVATE sagomacad_doc)
add_test(NAME expression COMMAND expression_test)
add_executable(command_test tests/command.cpp)
target_link_libraries(command_test PRIVATE sagomacad_engine)
add_test(NAME command COMMAND command_test)
add_executable(primitive_test tests/primitive.cpp)
target_link_libraries(primitive_test PRIVATE sagomacad_engine)
add_test(NAME primitive COMMAND primitive_test)
add_executable(regen_test tests/regen.cpp)
target_link_libraries(regen_test PRIVATE sagomacad_engine)
add_test(NAME regen COMMAND regen_test)
add_executable(camera_test tests/camera.cpp)
target_link_libraries(camera_test PRIVATE sagomacad_ui)
add_test(NAME camera COMMAND camera_test)
add_executable(export_stl_test tests/export_stl.cpp)
target_link_libraries(export_stl_test PRIVATE sagomacad_engine)
add_test(NAME export_stl COMMAND export_stl_test)
add_executable(parameter_editor_test tests/parameter_editor.cpp)
target_link_libraries(parameter_editor_test PRIVATE sagomacad_ui)
add_test(NAME parameter_editor COMMAND parameter_editor_test)
add_test(NAME fuzz_commands COMMAND sagomacad-cli fuzz-commands)
add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_cli.py
  $<TARGET_FILE:sagomacad> $<TARGET_FILE:sagomacad-cli>)
