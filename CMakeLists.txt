cmake_minimum_required(VERSION 3.25)
project(sagomacad VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()

find_package(SDL3 CONFIG REQUIRED)
find_package(imgui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(lunasvg CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(OpenCASCADE CONFIG REQUIRED)
find_path(STB_INCLUDE_DIRS stb_image_write.h PATH_SUFFIXES stb REQUIRED)
find_path(CLAY_INCLUDE_DIRS clay.h REQUIRED)

foreach(module core)
  add_library(sagomacad_${module} INTERFACE)
  target_include_directories(sagomacad_${module} INTERFACE ${PROJECT_SOURCE_DIR}/src)
endforeach()
add_library(sagomacad_doc STATIC src/doc/document.cpp src/doc/expression.cpp src/doc/sketch.cpp)
target_include_directories(sagomacad_doc PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(sagomacad_doc PUBLIC sagomacad_core nlohmann_json::nlohmann_json)

add_library(sagomacad_engine STATIC src/engine/registry.cpp src/engine/tools/model_preview.cpp src/engine/tools/sketch_tool.cpp src/engine/tools/sketch_overlay.cpp src/engine/cmds/document_cmds.cpp
    src/engine/cmds/param_cmds.cpp src/engine/cmds/edit_cmds.cpp src/engine/cmds/primitive_cmds.cpp src/engine/cmds/export_cmds.cpp src/engine/cmds/body_cmds.cpp src/engine/cmds/feature_cmds.cpp src/engine/cmds/measure_cmds.cpp src/engine/cmds/sketch_cmds.cpp src/engine/cmds/model_cmds.cpp)
target_include_directories(sagomacad_engine PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_compile_definitions(sagomacad_engine PRIVATE SAGOMACAD_EXPORT_DIR="${PROJECT_SOURCE_DIR}/export")
target_link_libraries(sagomacad_engine PUBLIC sagomacad_core sagomacad_doc sagomacad_solver sagomacad_kernel sagomacad_regen sagomacad_mesh)
add_library(sagomacad_automation STATIC src/automation/dispatch.cpp src/automation/mcp/server.cpp
  src/automation/control_client.cpp src/automation/session_file.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/docs/modeling-guide.md)
file(READ ${PROJECT_SOURCE_DIR}/docs/modeling-guide.md SAGOMACAD_MODELING_GUIDE)
configure_file(src/automation/mcp/modeling_guide.hpp.in ${CMAKE_BINARY_DIR}/generated/modeling_guide.hpp @ONLY)
target_include_directories(sagomacad_automation PRIVATE ${CMAKE_BINARY_DIR}/generated)
target_include_directories(sagomacad_automation PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_compile_definitions(sagomacad_automation PRIVATE SAGOMACAD_EXPORT_DIR="${PROJECT_SOURCE_DIR}/export")
target_link_libraries(sagomacad_automation PUBLIC sagomacad_engine nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(sagomacad_automation PRIVATE ws2_32)
endif()
target_link_libraries(sagomacad_ui PUBLIC sagomacad_engine)

add_library(sagomacad_shell STATIC apps/common/shell.cpp apps/common/sketch_widget.cpp apps/common/model_widget.cpp apps/common/control.cpp apps/common/view_cube_widget.cpp apps/common/timeline_widget.cpp)
target_include_directories(sagomacad_shell PUBLIC ${PROJECT_SOURCE_DIR}/apps/common PRIVATE ${STB_INCLUDE_DIRS})
file(COPY ${PROJECT_SOURCE_DIR}/assets/fonts ${PROJECT_SOURCE_DIR}/assets/icons DESTINATION ${CMAKE_BINARY_DIR}/assets)
target_compile_definitions(sagomacad_shell PRIVATE SAGOMACAD_ASSET_DIR="${CMAKE_BINARY_DIR}/assets")
target_link_libraries(sagomacad_shell PUBLIC sagomacad_ui sagomacad_automation SDL3::SDL3 imgui::imgui OpenGL::GL lunasvg::lunasvg nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(sagomacad_shell PRIVATE ws2_32)
endif()
add_executable(sagomacad apps/sagomacad/main.cpp)
target_link_libraries(sagomacad PRIVATE sagomacad_shell)
add_executable(sagomacad-cli apps/sagomacad-cli/main.cpp apps/sagomacad-cli/fuzz.cpp
  apps/sagomacad-cli/client.cpp apps/sagomacad-cli/mcp.cpp apps/sagomacad-cli/mcp_visual.cpp)
target_compile_definitions(sagomacad-cli PRIVATE SAGOMACAD_PROJECT_ROOT="${PROJECT_SOURCE_DIR}")
target_link_libraries(sagomacad-cli PRIVATE sagomacad_shell)
set_target_properties(sagomacad PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/apps/sagomacad$<0:>")
set_target_properties(sagomacad-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/apps/sagomacad-cli$<0:>")
